### Step 4 — Create `.claude/hooks/`

**`block-env-commit.sh`:**
```bash
#!/usr/bin/env bash
set -euo pipefail
input="$(cat)"
command="$(printf '%s' "$input" | sed -n 's/.*"command"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p')"
case "$command" in
  *"git add"*|*"git commit"*)
    if printf '%s' "$command" | grep -qE '(^|[[:space:]/])\.env(\.|$|[[:space:]])'; then
      echo "Blocked: command references .env — never commit env files." >&2
      exit 2
    fi
    ;;
esac
exit 0
```

**`prevent-tailwind-config.sh`:**
```bash
#!/usr/bin/env bash
set -euo pipefail
input="$(cat)"
file="$(printf '%s' "$input" | sed -n 's/.*"file_path"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p')"
case "$file" in
  */tailwind.config.js|*/tailwind.config.ts|*/tailwind.config.mjs|*/tailwind.config.cjs)
    echo "Blocked: Tailwind v4 uses @theme in src/app/globals.css — do NOT create tailwind.config.*" >&2
    exit 2
    ;;
esac
exit 0
```

**`lint-on-write.sh`:**
```bash
#!/usr/bin/env bash
set -euo pipefail
input="$(cat)"
file="$(printf '%s' "$input" | sed -n 's/.*"file_path"[[:space:]]*:[[:space:]]*"\([^"]*\)".*/\1/p')"
case "$file" in
  *.ts|*.tsx)
    if [ -f "$file" ]; then
      (cd "$(dirname "$0")/../.." && npx --no-install eslint "$file" 2>&1) || true
    fi
    ;;
esac
exit 0
```
